def reset_problem_tables():
    conn = get_conn()
    try:
        tables = [
            "signals",
            "trade_journal",
            "signal_participations",
            "followup_comments"
        ]

        for t in tables:
            conn.execute(f"DROP TABLE IF EXISTS {t}")
            print(f"[DROP] {t}")

        conn.commit()
    finally:
        conn.close()
